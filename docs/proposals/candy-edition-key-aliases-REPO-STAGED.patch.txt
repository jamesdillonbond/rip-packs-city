+// lib/candy/edition-key-aliases.ts
+//
+// #247 (2026-11-13): one Candy card was stored under several edition keys — the on-chain name is cut at 32
+// characters and the cut moves with the serial's digit count, a card hydrated from its off-chain JSON keys from
+// the full title, and older paths slugged punctuation differently. `candy_edition_key_aliases` maps each extra
+// key to the card's canonical key (refresh_candy_edition_key_aliases, daily), and DB triggers rewrite every
+// write that names an alias. The census resolves aliases HERE as well, before it compares a card's computed key
+// with the key its holding already has: otherwise every walk would "repoint" each card back to the alias (the
+// trigger then rewrites it again) — no data harm, but a repoint + listings/offers RPC per card per walk.
+//
+// Three answers, never two: a map (possibly empty) · the read failed (error — the caller fails the collection,
+// it does not walk without the map). A read at the row cap is a failure (it may be partial).
+
+export type AliasLabels = {
+  external_id: string
+  tier: string | null
+  player_name: string | null
+  team_name: string | null
+  set_name: string | null
+  circulation_count: number | null
+  thumbnail_url: string | null
+  video_url?: string | null
+  rarity_label?: string | null
+}
+
+export type EditionKeyAliases = {
+  /** alias key -> canonical key, only where the canonical edition's labels were read */
+  canonOf: Map<string, string>
+  /** canonical key -> its edition labels */
+  labels: Map<string, AliasLabels>
+}
+
+const CANDY_MLB_UUID = "209ade70-32c5-4470-bc7c-4793d660f713"
+const ROW_CAP = 1000
+const LABEL_COLS = "external_id,tier,player_name,team_name,set_name,circulation_count,thumbnail_url,video_url,rarity_label"
+const IN_CHUNK = 150
+
+// eslint-disable-next-line @typescript-eslint/no-explicit-any
+type Db = any
+
+export const NO_ALIASES: EditionKeyAliases = { canonOf: new Map(), labels: new Map() }
+
+export async function loadEditionKeyAliases(db: Db, collectionPrefix: string): Promise<{ aliases: EditionKeyAliases; error: string | null }> {
+  const { data, error } = await db.from("candy_edition_key_aliases")
+    .select("alias_key,canonical_key")
+    .eq("collection_prefix", collectionPrefix)
+    .limit(ROW_CAP)
+  if (error) return { aliases: NO_ALIASES, error: `edition key aliases read: ${error.message}` }
+  const rows = (data ?? []) as Array<{ alias_key: string; canonical_key: string }>
+  if (rows.length >= ROW_CAP) return { aliases: NO_ALIASES, error: `edition key aliases read: ${ROW_CAP}-row cap reached` }
+  if (!rows.length) return { aliases: NO_ALIASES, error: null }
+
+  const canonKeys = [...new Set(rows.map((r) => r.canonical_key))]
+  const labels = new Map<string, AliasLabels>()
+  for (let i = 0; i < canonKeys.length; i += IN_CHUNK) {
+    const { data: eds, error: eErr } = await db.from("editions")
+      .select(LABEL_COLS)
+      .eq("collection_id", CANDY_MLB_UUID)
+      .in("external_id", canonKeys.slice(i, i + IN_CHUNK))
+    if (eErr) return { aliases: NO_ALIASES, error: `canonical editions read: ${eErr.message}` }
+    for (const e of (eds ?? []) as AliasLabels[]) labels.set(e.external_id, e)
+  }
+  const canonOf = new Map<string, string>()
+  // an alias whose canonical edition is missing is left alone (the card keeps its computed key)
+  for (const r of rows) if (labels.has(r.canonical_key)) canonOf.set(r.alias_key, r.canonical_key)
+  return { aliases: { canonOf, labels }, error: null }
+}

+CREATE OR REPLACE FUNCTION public.refresh_candy_edition_key_aliases()
+returns jsonb language plpgsql security definer set search_path = public, pg_temp as $fn$
+declare
+  c_candy constant uuid := '209ade70-32c5-4470-bc7c-4793d660f713';
+  v_started timestamptz := clock_timestamp();
+  v_groups int := 0;
+  v_new int := 0;
+begin
+  drop table if exists pg_temp._ek;
+  drop table if exists pg_temp._grp;
+  create temp table _ek on commit drop as
+  select ed.external_id, split_part(ed.external_id, ':', 2) as pfx, ed.tier, ed.circulation_count, ed.thumbnail_url
+    from public.editions ed
+   where ed.collection_id = c_candy
+     and ed.external_id like 'c:%'
+     and ed.thumbnail_url is not null
+     and coalesce(ed.circulation_count, 0) > 1
+     and ed.external_id !~ ':sealed-item';
+
+  create temp table _grp on commit drop as
+  select pfx, tier, circulation_count, thumbnail_url, array_agg(external_id) as keys
+    from _ek
+   group by pfx, tier, circulation_count, thumbnail_url
+  having count(*) > 1;
+  get diagnostics v_groups = row_count;
+
+  insert into public.candy_edition_key_aliases (alias_key, canonical_key, collection_prefix)
+  select k, p.canon, 'c:' || p.pfx
+    from (
+      select g.pfx, g.keys,
+             coalesce(
+               -- a member that is already canonical keeps the group's canonical stable
+               (select k0 from unnest(g.keys) k0
+                 where exists (select 1 from public.candy_edition_key_aliases a where a.canonical_key = k0)
+                 order by k0 limit 1),
+               (select k1 from unnest(g.keys) k1
+                 order by (select count(*) from public.wallet_moments_cache w
+                            where w.collection_id = c_candy and w.edition_key = k1) desc,
+                          length(k1) desc, k1
+                 limit 1)
+             ) as canon
+        from _grp g
+    ) p
+    cross join lateral unnest(p.keys) k
+   where k <> p.canon
+     -- a canonical is never made an alias (no chains)
+     and not exists (select 1 from public.candy_edition_key_aliases a where a.canonical_key = k)
+  on conflict (alias_key) do nothing;
+  get diagnostics v_new = row_count;
+
+  perform public.log_pipeline_run(

+      join public.candy_edition_key_aliases a on a.alias_key = w.edition_key
+     where w.collection_id = c_candy
+     limit greatest(coalesce(p_limit, 5000), 0)
+  )
+  update public.wallet_moments_cache w
+     set edition_key = b.canonical_key,
+         tier        = coalesce(ce.tier, w.tier),
+         mint_count  = coalesce(ce.circulation_count, w.mint_count),
+         set_name    = coalesce(ce.set_name, w.set_name)
+    from batch b
+    join public.editions ce on ce.collection_id = c_candy and ce.external_id = b.canonical_key
+   where w.id = b.id;
+  get diagnostics v_wmc = row_count;
+
+  -- market rows: few, moved whole (the edition_id triggers do the same on every new write)
+  with m as (
+    select ae.id as alias_id, ce.id as canon_id
+      from public.candy_edition_key_aliases a
+      join public.editions ae on ae.collection_id = c_candy and ae.external_id = a.alias_key
+      join public.editions ce on ce.collection_id = c_candy and ce.external_id = a.canonical_key
+  )
+  update public.sales s set edition_id = m.canon_id
+    from m where s.collection_id = c_candy and s.edition_id = m.alias_id;
+  get diagnostics v_sales = row_count;
+
+  with m as (
+    select ae.id as alias_id, ce.id as canon_id
+      from public.candy_edition_key_aliases a
+      join public.editions ae on ae.collection_id = c_candy and ae.external_id = a.alias_key
+      join public.editions ce on ce.collection_id = c_candy and ce.external_id = a.canonical_key
+  )
+  update public.candy_listings l set edition_id = m.canon_id from m where l.edition_id = m.alias_id;
+  get diagnostics v_listings = row_count;
+
+  with m as (
+    select ae.id as alias_id, ce.id as canon_id
+      from public.candy_edition_key_aliases a
+      join public.editions ae on ae.collection_id = c_candy and ae.external_id = a.alias_key
+      join public.editions ce on ce.collection_id = c_candy and ce.external_id = a.canonical_key
+  )
+  update public.candy_offers o set edition_id = m.canon_id from m where o.edition_id = m.alias_id;
+  get diagnostics v_offers = row_count;
+
+  select count(*) into v_left
+    from public.wallet_moments_cache w
+    join public.candy_edition_key_aliases a on a.alias_key = w.edition_key
+   where w.collection_id = c_candy;
+
+  perform public.log_pipeline_run(
+    'candy-edition-key-alias-apply', v_started, v_wmc + v_sales + v_listings + v_offers,
+    v_wmc + v_sales + v_listings + v_offers, 0, true, null, 'candy_mlb', null, null,
+    jsonb_build_object('wmc_moved', v_wmc, 'sales_moved', v_sales, 'listings_moved', v_listings,
+                       'offers_moved', v_offers, 'wmc_left', v_left, 'limit', p_limit));
+  return jsonb_build_object('ok', true, 'wmc_moved', v_wmc, 'sales_moved', v_sales, 'listings_moved', v_listings,
+                            'offers_moved', v_offers, 'wmc_left', v_left);
+end $fn$;
