import { createClient } from "https://esm.sh/@supabase/supabase-js@2.45.0"

const FLOW_REST = "https://rest-mainnet.onflow.org"
const GOLAZOS_COLLECTION_ID = "06248cc4-b85f-47cd-af67-1855d14acd75"

const SUPABASE_URL = Deno.env.get("SUPABASE_URL") ?? Deno.env.get("NEXT_PUBLIC_SUPABASE_URL")!
const SERVICE_KEY = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY") ?? Deno.env.get("SERVICE_ROLE_KEY")!
const supabase = createClient(SUPABASE_URL, SERVICE_KEY)

function b64ToUtf8(b64: string): string {
  const binary = atob(b64)
  const bytes = new Uint8Array(binary.length)
  for (let i = 0; i < binary.length; i++) bytes[i] = binary.charCodeAt(i)
  return new TextDecoder("utf-8").decode(bytes)
}

function unwrap(node: any): any {
  if (node === null || node === undefined) return node
  if (Array.isArray(node)) return node.map(unwrap)
  if (typeof node !== "object") return node
  const { type, value } = node
  if (type !== undefined && value !== undefined) {
    switch (type) {
      case "Optional": return value === null ? null : unwrap(value)
      case "Array": return (value as any[]).map(unwrap)
      case "Dictionary": {
        const o: Record<string, any> = {}
        for (const kv of value as any[]) o[String(unwrap(kv.key))] = unwrap(kv.value)
        return o
      }
      case "Struct": case "Resource": case "Event": case "Contract": case "Enum": {
        const o: Record<string, any> = {}
        for (const f of (value.fields ?? [])) o[f.name] = unwrap(f.value)
        return o
      }
      default: return value
    }
  }
  return node
}

async function runCadence(code: string, args: Array<{ type: string; value: unknown }> = []): Promise<any> {
  const body = { script: btoa(code), arguments: args.map(a => btoa(JSON.stringify(a))) }
  const res = await fetch(`${FLOW_REST}/v1/scripts?block_height=sealed`, {
    method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(body),
  })
  const text = await res.text()
  if (!res.ok) throw new Error(`Flow HTTP ${res.status}: ${text.slice(0, 300)}`)
  return unwrap(JSON.parse(b64ToUtf8(JSON.parse(text))))
}

const IDS_SCRIPT = `
import Golazos from 0x87ca73a41bb50ad5
import NonFungibleToken from 0x1d7e57aa55817448
access(all) fun main(addr: Address): [UInt64] {
  let acct = getAccount(addr)
  let ref = acct.capabilities.borrow<&{NonFungibleToken.CollectionPublic}>(Golazos.CollectionPublicPath)
  if ref == nil { return [] }
  return ref!.getIDs()
}`

const BATCH_META_SCRIPT = `
import Golazos from 0x87ca73a41bb50ad5
import NonFungibleToken from 0x1d7e57aa55817448
access(all) fun main(addr: Address, ids: [UInt64]): [{String: UInt64}] {
  let acct = getAccount(addr)
  let ref = acct.capabilities.borrow<&{NonFungibleToken.CollectionPublic}>(Golazos.CollectionPublicPath)
  if ref == nil { return [] }
  let out: [{String: UInt64}] = []
  for id in ids {
    if let nft = ref!.borrowNFT(id) {
      let g = nft as! &Golazos.NFT
      out.append({ "id": g.id, "editionID": g.editionID, "serialNumber": g.serialNumber })
    }
  }
  return out
}`

Deno.serve(async (req: Request) => {
  const auth = req.headers.get("Authorization")
  if (auth !== "Bearer __REDACTED_SECRET_2__") return new Response("Unauthorized", { status: 401 })

  const url = new URL(req.url)
  const wallet = url.searchParams.get("wallet")
  if (!wallet || !wallet.startsWith("0x")) {
    return new Response(JSON.stringify({ error: "wallet param required" }), { status: 400 })
  }

  const started = Date.now()
  try {
    const ids = await runCadence(IDS_SCRIPT, [{ type: "Address", value: wallet }]) as string[]
    if (!Array.isArray(ids) || ids.length === 0) {
      return new Response(JSON.stringify({ ok: true, wallet, found: 0, upserted: 0 }), {
        headers: { "Content-Type": "application/json; charset=utf-8" },
      })
    }

    const moments: Array<{ id: string; editionID: string; serialNumber: string }> = []
    for (let i = 0; i < ids.length; i += 50) {
      const chunk = ids.slice(i, i + 50)
      const meta = await runCadence(BATCH_META_SCRIPT, [
        { type: "Address", value: wallet },
        { type: "Array", value: chunk.map(id => ({ type: "UInt64", value: String(id) })) },
      ]) as Array<{ id: string; editionID: string; serialNumber: string }>
      for (const m of meta) moments.push(m)
      if (Date.now() - started > 45_000) break
    }

    const editionIds = [...new Set(moments.map(m => String(m.editionID)))]
    const edMap = new Map<string, any>()
    for (let i = 0; i < editionIds.length; i += 500) {
      const batch = editionIds.slice(i, i + 500)
      const { data } = await supabase
        .from("editions")
        .select("external_id, player_name, set_name, tier, series")
        .eq("collection_id", GOLAZOS_COLLECTION_ID)
        .in("external_id", batch)
      for (const row of data ?? []) edMap.set(String((row as any).external_id), row)
    }

    const now = new Date().toISOString()
    const rows = moments.map(m => {
      const ed = edMap.get(String(m.editionID)) ?? {}
      return {
        wallet_address: wallet.toLowerCase(),
        moment_id: String(m.id),
        edition_key: String(m.editionID),
        serial_number: Number(m.serialNumber),
        player_name: ed.player_name ?? null,
        set_name: ed.set_name ?? null,
        tier: ed.tier ?? null,
        series_number: ed.series ?? null,
        collection_id: GOLAZOS_COLLECTION_ID,
        last_seen_at: now,
      }
    })

    let upserted = 0
    for (let i = 0; i < rows.length; i += 100) {
      const chunk = rows.slice(i, i + 100)
      const { error } = await supabase
        .from("wallet_moments_cache")
        .upsert(chunk, { onConflict: "wallet_address,moment_id" })
      if (error) console.log("[scan-golazos] upsert err:", error.message)
      else upserted += chunk.length
    }

    return new Response(JSON.stringify({
      ok: true, wallet, found: moments.length, upserted, elapsed: Date.now() - started,
    }), { headers: { "Content-Type": "application/json; charset=utf-8" } })
  } catch (err: any) {
    console.log("[scan-golazos] fatal:", err.message)
    return new Response(JSON.stringify({ ok: false, error: err.message }), {
      status: 500, headers: { "Content-Type": "application/json; charset=utf-8" },
    })
  }
})
